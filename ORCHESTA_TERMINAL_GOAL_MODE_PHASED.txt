# ============================================================
# 2. TERMINAL STATE MACHINE
# ============================================================
#
# Implement an explicit terminal state machine.
#
# At minimum:
#
# enum TerminalInteractionMode {
#   ShellPrompt,
#   CommandRunning,
#   InteractiveProgram,
#   AlternateScreen,
#   PasswordInput,
#   ShellFallback,
#   Disconnected,
# }
#
#
# ------------------------------------------------------------
# ShellPrompt
# ------------------------------------------------------------
#
# The shell is ready for a new command.
#
# Keyboard input belongs to Orchesta's CommandEditor.
#
# DO NOT send ordinary typing directly to the PTY.
#
# The user must be able to:
#
# - click anywhere inside the command
# - position caret with mouse
# - drag-select text
# - double click a word
# - triple click a logical line
# - select ranges
# - type over selection
# - insert/delete in the middle
# - use multiline commands
# - use Shift+Enter for newline
# - use Ctrl+Enter optionally for newline
# - use Alt+Enter optionally for newline
# - use Home/End
# - Ctrl+A / Ctrl+E
# - Ctrl+Left / Ctrl+Right
# - Alt+Left / Alt+Right
# - word selection
# - subword movement
# - undo
# - redo
# - copy
# - cut
# - paste
# - multiline paste
# - autocomplete
# - history
# - syntax highlighting
# - bracket matching
# - auto-close pairs
# - multiple cursors
#
#
# ------------------------------------------------------------
# CommandRunning
# ------------------------------------------------------------
#
# As soon as Enter executes the editor buffer:
#
# editor input is no longer the owner of keyboard input.
#
# Keyboard events MUST be capable of going to the PTY.
#
# This is essential for:
#
# sudo
# prompts
# yes/no questions
# readline
# interactive installers
# package managers
# CLIs
#
#
# ------------------------------------------------------------
# InteractiveProgram / AlternateScreen
# ------------------------------------------------------------
#
# For:
#
# vim
# nvim
# htop
# btop
# less
# nano
# Claude Code
# Codex
# Gemini CLI
# OpenCode
# etc.
#
# Orchesta MUST behave like a proper terminal.
#
# The RichCommandEditor must NOT intercept:
#
# arrows
# Enter
# Escape
# Tab
# Ctrl+C
# Ctrl+D
# mouse reports
# bracketed paste
# function keys
# Kitty keyboard sequences
# application cursor mode
#
# Send them to PTY according to terminal protocol/state.
#

# 3. DO NOT DETECT TUIs BY PROCESS NAME ONLY
# ============================================================
#
# Do not create fragile code like:
#
# if process === "vim" || process === "claude"
#
# Process names can provide auxiliary hints but must NOT be the primary
# architecture.
#
# Detect terminal state using a combination of:
#
# - shell preexec/precmd lifecycle
# - PTY foreground process group where available
# - alternate-screen DEC modes
# - terminal control sequences
# - mouse-reporting modes
# - application cursor-key mode
# - shell integration
# - prompt markers
# - command-running state
#
# The primary simple rule should be:
#
# At a confirmed shell prompt:
#     RichCommandEditor owns text entry.
#
# After command execution and until the next confirmed shell prompt:
#     PTY owns keyboard interaction.
#
# This naturally supports most CLI/TUI programs.

# ============================================================
#
# Enter:
#     execute command
#
# Shift+Enter:
#     insert newline
#
# Optionally:
#
# Ctrl+Enter
# Alt+Enter
#

#

#

#

# ============================================================
# 7. SOFT WRAPPING
# ============================================================
#
# Long commands should wrap visually without inserting newline characters.
#
# Differentiate:
#
# logical lines
#
# from:
#
# visual wrapped lines.
#
# Selection and cursor movement must understand the difference.
#
# Resize must rewrap the editor without mutating its underlying text.

#

# ''

# []
# {}
